\section{实验原理}

\subsection{抽样过程与频谱分析}
离散信号可由连续信号 $F(t)$ 经周期性矩形窄脉冲 $S(t)$ 抽样获得。设抽样间隔为 $T_s$，抽样频率 $f_s = 1/T_s$，则抽样信号 $F_s(t)$ 可表示为：
\begin{equation}
    F_s(t) = F(t) \cdot S(t)
\end{equation}

在频域上，连续周期信号经抽样后的频谱 $F_s(j\omega)$ 是原信号频谱的周期性延拓：
\begin{equation}
    F_s(j\omega) = \frac{A\tau}{T} \sum_{m=-\infty}^{+\infty} \text{Sa}\left(\frac{m\omega_s\tau}{2}\right) F(\omega - m\omega_s)
\end{equation}
其中，频谱重复周期为 $f_s$，幅度受抽样脉冲的傅里叶系数加权。这意味着抽样信号的频谱包含了原信号频谱以及以 $f_s$ 为间隔的无限个复制分量。

\subsection{抽样定理与信号恢复}
要从抽样信号中无失真地恢复出原信号，必须满足**香农抽样定理**：
\begin{equation}
    f_s \geq 2B_f
\end{equation}
其中 $B_f$ 为原信号的最高频率分量。
\begin{itemize}
    \item 若 $f_s \geq 2B_f$：频谱不发生混叠，通过截止频率 $f_c$ ($f_m \leq f_c \leq f_s - f_m$) 的低通滤波器即可恢复原信号。
    \item 若 $f_s < 2B_f$：频谱发生混叠（Aliasing），无法通过低通滤波器恢复原信号。
\end{itemize}
工程实际中，为减小滤波器非理想特性带来的失真，通常采用过采样，取 $f_s > 2.56 f_m$。

\end{enumerate}

\section{实验结论}

\subsection{同步采样与异步采样的特点}
\begin{enumerate}
    \item \textbf{同步采样}：采样时钟与信号源同源，时间间隔严格固定。其优点是便于后续处理，适合周期性信号，能有效避免非相干采样带来的频谱泄漏；缺点是系统对时钟同步要求高。
    \item \textbf{异步采样}：采样时钟与信号源不同源。其灵活性高，适用于信号变化快或不规则的场景；但由于采样点相对于信号周期的位置不固定，容易引入额外的相位噪声，且在重建时处理较为复杂。
\end{enumerate}

\subsection{抽样率与滤波器的影响}
\begin{enumerate}
    \item \textbf{抽样率}：直接决定了信号是否发生混叠。实验验证了当 $f_s < 2f_m$ 时，高频成分折叠到低频段，造成不可逆的失真；提高抽样率（过采样）能降低对滤波器陡峭度的要求，提升恢复质量。
    \item \textbf{滤波器}：低通滤波器用于滤除抽样产生的高频延拓分量。滤波器的截止频率必须介于 $f_m$ 和 $f_s - f_m$ 之间。实际滤波器的过渡带非理想特性要求我们在工程上采用 $f_s > 2.56 f_m$ 的过采样策略。
\end{enumerate}

\textbf{题目}：利用 MATLAB 仿真验证抽样定理。产生一个频率为 $100\,\text{Hz}$ 的正弦信号，分别用 $150\,\text{Hz}$（欠采样）和 $500\,\text{Hz}$（过采样）进行抽样，并观察恢复效果。

\begin{lstlisting}[caption={抽样定理验证 MATLAB 代码}]
% 参数设置
fm = 100;           % 信号频率 100Hz
T = 0.1;            % 信号持续时间
t = 0:0.0001:T;     % 模拟连续时间轴
xt = sin(2*pi*fm*t); % 原始信号

% 1. 欠采样 (fs = 150Hz < 2*fm)
fs1 = 150;
ts1 = 0:1/fs1:T;
xn1 = sin(2*pi*fm*ts1);

% 2. 过采样 (fs = 500Hz > 2*fm)
fs2 = 500;
ts2 = 0:1/fs2:T;
xn2 = sin(2*pi*fm*ts2);

% 绘图对比
figure;
subplot(2,1,1);
plot(t, xt, 'b'); hold on;
stem(ts1, xn1, 'r');
title('欠采样 fs=150Hz (混叠)');
legend('原信号', '抽样点');

subplot(2,1,2);
plot(t, xt, 'b'); hold on;
stem(ts2, xn2, 'r');
title('过采样 fs=500Hz (无失真)');
legend('原信号', '抽样点');
\end{lstlisting}
